<script>
const initialCategory = <?= json_encode($current_category) ?>;
const projects = <?= json_encode($projects_array, JSON_UNESCAPED_UNICODE) ?>;

let currentProject = null;
let currentSlide = 0;

const projectsGrid = document.getElementById('projectsGrid');
const modalOverlay = document.getElementById('modalOverlay');
const modalClose = document.getElementById('modalClose');
const sliderTrack = document.getElementById('sliderTrack');
const sidebarLinks = document.querySelectorAll('.sidebar-link');
const sidebar = document.getElementById('sidebar');
const burgerMenu = document.querySelector('.burger-menu');
const modalSliderWrapper = document.getElementById('modalSliderWrapper');

function toggleSidebar() {
    sidebar.classList.toggle('open');
    // Анимация бургера можно добавить здесь
}

// Закрытие сайдбара при клике вне его на мобильных
document.addEventListener('click', (e) => {
    if (window.innerWidth <= 1024 && !sidebar.contains(e.target) && !burgerMenu.contains(e.target) && sidebar.classList.contains('open')) {
        sidebar.classList.remove('open');
    }
});

function getLayoutClass(count) {
    if (count === 1) return 'layout-1';
    if (count === 2) return 'layout-2';
    if (count === 3) return 'layout-3';
    return 'layout-4'; // Для 4 и более упрощенно
}

function renderProjects(category) {
    projectsGrid.innerHTML = '';
    const filtered = category === 'all' 
        ? projects 
        : projects.filter(p => p.category === category);

    if (filtered.length === 0) {
        projectsGrid.innerHTML = '<p style="grid-column: 1/-1; text-align: center; color: var(--gray); padding: 50px;">В этой категории пока нет работ.</p>';
        return;
    }

    filtered.forEach(project => {
        const card = document.createElement('div');
        card.className = 'project-card';
        card.onclick = () => openModal(project);
        
        const imagesCount = project.images.length;
        const layoutClass = getLayoutClass(imagesCount > 4 ? 4 : imagesCount);
        
        // Берем первые 4 фото для превью
        const previewImages = project.images.slice(0, 4);
        
        let imagesHTML = '';
        previewImages.forEach((img, index) => {
            imagesHTML += `<div class="img-item"><img src="${img}" alt="Photo ${index + 1}"></div>`;
        });
        
        card.innerHTML = `
        <div class="project-images ${layoutClass}">${imagesHTML}</div>
        <div class="project-info">
            <div class="project-client">${project.client}</div>
            <div class="project-desc">${project.description || 'Фотографическая история...'}</div>
        </div>`;
        projectsGrid.appendChild(card);
    });
}

// Фильтрация
sidebarLinks.forEach(link => {
    link.addEventListener('click', () => {
        sidebarLinks.forEach(l => l.classList.remove('active'));
        link.classList.add('active');
        if (window.innerWidth <= 1024) sidebar.classList.remove('open');
        
        projectsGrid.classList.add('fade-out');
        setTimeout(() => {
            renderProjects(link.dataset.category);
            history.replaceState(null, '', '?category=' + link.dataset.category);
            setTimeout(() => projectsGrid.classList.remove('fade-out'), 50);
        }, 300);
    });
});

renderProjects(initialCategory);

// Modal Logic
function openModal(project) {
    currentProject = project;
    currentSlide = 0;
    document.getElementById('modalClient').textContent = project.client;
    document.getElementById('modalDescription').textContent = project.description || '';
    renderSlider();
    modalOverlay.classList.add('active');
    document.body.style.overflow = 'hidden';
}

function closeModal() {
    modalOverlay.classList.remove('active');
    document.body.style.overflow = '';
}

function renderSlider() {
    if (!currentProject) return;
    sliderTrack.innerHTML = currentProject.images.map(img => `
        <div class="slider-slide"><img src="${img}" alt="Project Image" loading="lazy"></div>
    `).join('');
    updateSlider();
}

function updateSlider() {
    sliderTrack.style.transform = `translateX(-${currentSlide * 100}%)`;
}

function nextSlide() { 
    if(!currentProject) return;
    currentSlide = (currentSlide + 1) % currentProject.images.length; 
    updateSlider(); 
}

function prevSlide() { 
    if(!currentProject) return;
    currentSlide = (currentSlide - 1 + currentProject.images.length) % currentProject.images.length; 
    updateSlider(); 
}

// Events
modalClose.addEventListener('click', closeModal);
modalOverlay.addEventListener('click', (e) => { if (e.target === modalOverlay) closeModal(); });
document.getElementById('sliderNext').addEventListener('click', (e) => { e.stopPropagation(); nextSlide(); });
document.getElementById('sliderPrev').addEventListener('click', (e) => { e.stopPropagation(); prevSlide(); });

// Keyboard & Swipe
document.addEventListener('keydown', (e) => {
    if (!modalOverlay.classList.contains('active')) return;
    if (e.key === 'Escape') closeModal();
    if (e.key === 'ArrowRight') nextSlide();
    if (e.key === 'ArrowLeft') prevSlide();
});

let touchStartX = 0;
modalSliderWrapper.addEventListener('touchstart', e => touchStartX = e.changedTouches[0].screenX, {passive: true});
modalSliderWrapper.addEventListener('touchend', e => {
    const diff = touchStartX - e.changedTouches[0].screenX;
    if (Math.abs(diff) > 50) diff > 0 ? nextSlide() : prevSlide();
}, {passive: true});

</script>
